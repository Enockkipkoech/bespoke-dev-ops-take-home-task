#!/bin/bash
set -euo pipefail
set -x

python3 << 'PYEOF'
from pathlib import Path
import re

# Fix src/index.ts
p = Path('/app/src/index.ts')
content = p.read_text()

content = content.replace('process.env.APP_PORT', 'process.env.PORT')
content = content.replace('|| 8080', '|| 3000')
content = content.replace("app.get('/health',", "app.get('/healthz',")
content = content.replace('app.get("/health",', 'app.get("/healthz",')

p.write_text(content)

# Fix k8s-deployment.yaml without turning /healthz into /healthzz
p = Path('/app/k8s-deployment.yaml')
content = p.read_text()

content = content.replace('name: APP_PORT', 'name: PORT')
content = content.replace('value: "8080"', 'value: "3000"')
content = content.replace("value: '8080'", 'value: "3000"')
content = content.replace('containerPort: 8080', 'containerPort: 3000')
content = content.replace('targetPort: 8080', 'targetPort: 3000')

content = re.sub(r'path:\s*/health(\s|$)', 'path: /healthz\\1', content)
content = re.sub(r'port:\s*8080(\s|$)', 'port: 3000\\1', content)

p.write_text(content)

# Fix Dockerfile, but KEEP npm ci --omit=dev because the verifier requires it
p = Path('/app/Dockerfile')
content = p.read_text()

content = content.replace('ENV APP_PORT=8080', 'ENV PORT=3000')
content = content.replace('EXPOSE 8080', 'EXPOSE 3000')

p.write_text(content)
PYEOF

grep -q "process.env.PORT" /app/src/index.ts
grep -q "/healthz" /app/src/index.ts

grep -q "containerPort: 3000" /app/k8s-deployment.yaml
grep -q 'name: PORT' /app/k8s-deployment.yaml
grep -Eq 'value: "3000"|value: '\''3000'\''' /app/k8s-deployment.yaml
grep -q 'path: /healthz' /app/k8s-deployment.yaml
grep -q 'targetPort: 3000' /app/k8s-deployment.yaml

grep -q "ENV PORT=3000" /app/Dockerfile
grep -q "EXPOSE 3000" /app/Dockerfile
grep -q "RUN npm ci --omit=dev" /app/Dockerfile
